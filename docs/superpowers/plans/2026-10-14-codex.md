markdown.ts (base):

```ts
import { marked } from "marked";

marked.setOptions({ breaks: true, mangle: false, headerIds: false } as never);

const ALLOWED = new Set([
  "P", "H1", "H2", "H3", "H4", "H5", "H6", "UL", "OL", "LI",
  "BLOCKQUOTE", "CODE", "PRE", "EM", "STRONG", "B", "I", "A", "BR", "IMG",
]);
const KEEP_ATTRS: Record<string, Set<string>> = {
  A: new Set(["title"]),
  IMG: new Set(["src", "alt"]),
};

export function renderMarkdown(body: string): string {
  let html: string;
  try {
    html = marked.parse(body ?? "", { async: false }) as string;
  } catch {
    return escapeHtml(body ?? "");
  }
  const doc = new DOMParser().parseFromString(html, "text/html");
  return sanitize(...doc.body.childNodes);
  // sanitize walks: disallowed tag → replaced by its textContent fragments;
  // allowed: attributes filtered to KEEP_ATTRS; IMG src without ^data:image/ DROPPED (whole img replaced by alt text); A href: DROPPED (keep text, keep title).
}
```

Final implementation is the walk above — no innerHTML round-trip; escapeHtml helper for the catch path. Tests: h1/strong/code/paragraph render; `<script>alert(1)</script>` → text; `onerror` attr stripped; `a href` dropped but text kept; img data URL kept, external src dropped.
