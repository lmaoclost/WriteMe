```ts
export interface Pt { x: number; y: number }

export function polygonArea(pts: Pt[]): number {
  let a = 0;
  for (let i = 0, j = pts.length - 1; i < pts.length; j = i++) {
    a += (pts[j].x + pts[i].x) * (pts[j].y - pts[i].y);
  }
  return Math.abs(a / 2);
}

export function isMeaningful(pts: Pt[]): boolean {
  return polygonArea(pts) > 1e-5;
}

export function pointInPolygon(pt: Pt, pts: Pt[]): boolean {
  let inside = false;
  for (let i = 0, j = pts.length - 1; i < pts.length; j = i++) {
    const xi = pts[i].x, yi = pts[i].y, xj = pts[j].x, yj = pts[j].y;
    const intersect = yi > pt.y !== yj > pt.y && pt.x < ((xj - xi) * (pt.y - yi)) / (yj - yi) + xi;
    if (intersect) inside = !inside;
  }
  return inside;
}

export function regionCentroid(pts: Pt[]): Pt {
  const n = pts.length;
  if (n < 3) {
    const s = pts.reduce((acc, p) => ({ x: acc.x + p.x, y: acc.y + p.y }), { x: 0, y: 0 });
    return { x: s.x / (n || 1), y: s.y / (n || 1) };
  }
  let cross = 0, cx = 0, cy = 0;
  for (let i = 0, j = n - 1; i < n; j = i++) {
    const f = pts[j].x * pts[i].y - pts[i].x * pts[j].y;
    cross += f;
    cx += (pts[j].x + pts[i].x) * f;
    cy += (pts[j].y + pts[i].y) * f;
  }
  if (Math.abs(cross) < 1e-9) {
    let sx = 0, sy = 0;
    for (const p of pts) { sx += p.x; sy += p.y; }
    return { x: sx / n, y: sy / n };
  }
  return { x: cx / (3 * cross), y: cy / (3 * cross) };
}

export function projectToBoundary(pt: Pt, pts: Pt[]): { dist: number; point: Pt; edgeIndex: number; t: number } {
  let best = { dist: Infinity, point: { x: 0, y: 0 }, edgeIndex: 0, t: 0 };
  for (let i = 0; i < pts.length; i++) {
    const a = pts[i], b = pts[(i + 1) % pts.length];
    const dx = b.x - a.x, dy = b.y - a.y;
    const len2 = dx * dx + dy * dy;
    const t = len2 === 0 ? 0 : Math.max(0, Math.min(1, ((pt.x - a.x) * dx + (pt.y - a.y) * dy) / len2));
    const p = { x: a.x + t * dx, y: a.y + t * dy };
    const d = Math.hypot(pt.x - p.x, pt.y - p.y);
    if (d < best.dist) best = { dist: d, point: p, edgeIndex: i, t };
  }
  return best;
}

export interface SnapTarget { point: Pt; regionId: string }

export function snapToCarveBorder(
  pt: Pt,
  regions: { points: Pt[]; id: string }[],
  scale: number,
  opts?: { vertexPx?: number; edgePx?: number },
): SnapTarget | null {
  if (!scale) return null;
  const vertexPx = opts?.vertexPx ?? 14;
  const edgePx = opts?.edgePx ?? 8;
  let vertexHit: { dist: number; point: Pt; regionId: string } | null = null;
  let edgeHit: { dist: number; point: Pt; regionId: string; edgeIdx: number; t: number } | null = null;
  for (const r of regions) {
    if (r.points.length < 3) continue;
    for (const v of r.points) {
      const d = Math.hypot((pt.x - v.x), (pt.y - v.y)) * scale;
      if (d < (vertexHit?.dist ?? Infinity) && d <= vertexPx) vertexHit = { dist: d, point: { ...v }, regionId: r.id };
    }
    const p = projectToBoundary(pt, r.points);
    const d = p.dist * scale;
    if (d < (edgeHit?.dist ?? Infinity) && d <= edgePx) edgeHit = { dist: d, point: p.point, regionId: r.id, edgeIdx: p.edgeIndex, t: p.t };
  }
  if (vertexHit) return { point: vertexHit.point, regionId: vertexHit.regionId };
  if (edgeHit) return { point: edgeHit.point, regionId: edgeHit.regionId };
  return null;
}
```

```ts
import type { Pt } from "./geometry";
import { pointInPolygon, polygonArea } from "./geometry";

interface SnapRef { point: Pt; edgeIndex: number; t: number }

export function buildCarveChild(parentPts: Pt[], entry: SnapRef, exit: SnapRef, forward: boolean, middle: Pt[]): Pt[] {
  const n = parentPts.length;
  const i = entry.edgeIndex + entry.t;
  const o = exit.edgeIndex + exit.t;
  const span = ((i - o) % n + n) % n;
  const child: Pt[] = [{ x: entry.point.x, y: entry.point.y }, ...middle.map(p => ({ x: p.x, y: p.y })), { x: exit.point.x, y: exit.point.y }];
  const picks: { k: number; key: number }[] = [];
  for (let u = 0; u < n; u++) {
    if (forward) {
      const f = ((u - o) % n + n) % n;
      if (f > 1e-9 && f < span - 1e-9) picks.push({ k: u, key: f });
    } else {
      const f = ((o - u) % n + n) % n;
      if (f > 1e-9 && f < n - span - 1e-9) picks.push({ k: u, key: f });
    }
  }
  picks.sort((a, b) => a.key - b.key);
  for (const p of picks) child.push({ x: parentPts[p.k].x, y: parentPts[p.k].y });
  return child;
}

const key = (p: Pt) => `${Math.round(p.x / 1e-6)}_${Math.round(p.y / 1e-6)}`;
const pairKey = (a: string, b: string) => (a < b ? `${a}|${b}` : `${b}|${a}`);

export function unionBorderRing(regions: Pt[][]): { ring: Pt[]; ok: boolean } {
  const edgeCount = new Map<string, number>();
  const vertexOf = new Map<string, Pt>();
  for (const pts of regions) {
    for (let k = 0; k < pts.length; k++) {
      const a = pts[k], b = pts[(k + 1) % pts.length];
      const ka = key(a), kb = key(b);
      if (ka === kb) continue;
      vertexOf.set(ka, a);
      vertexOf.set(kb, b);
      const pk = pairKey(ka, kb);
      edgeCount.set(pk, (edgeCount.get(pk) || 0) + 1);
    }
  }
  // adjacency from unique (count==1) edges
  const adj = new Map<string, { to: string; pt: Pt }>();
  for (const [pk, count] of edgeCount) {
    if (count !== 1) continue;
    const [a, b] = pk.split("|");
    (adj.get(a) ?? adj.set(a, []).get(a)!).push({ to: b, pt: vertexOf.get(b)! });
    (adj.get(b) ?? adj.set(b, []).get(b)!).push({ to: a, pt: vertexOf.get(a)! });
  }
  const startKey = adj.keys().next().value;
  if (!startKey) return { ring: [], ok: false };
  const ring: Pt[] = [];
  const walked = new Set<string>([startKey]);
  let cur = startKey;
  // find one outgoing unique edge from start
  const first = adj.get(cur)![0];
  ring.push(vertexOf.get(cur)!, first.pt);
  cur = first.to;
  while (cur !== startKey) {
    if (walked.has(cur)) return { ring: [], ok: false }; // multiple cycles/islands
    walked.add(cur);
    const next = (adj.get(cur) || []).find(e => !walked.has(e.to)) ?? (cur === walked.values().next().value ? null : null);
    if (!next) break;
    ring.push(next.pt);
    cur = next.to;
    if (ring.length > 100000) return { ring: [], ok: false };
  }
  return { ring, ok: ring.length >= 3 };
}

export function detectCarveParent(pt: Pt, regions: { id: string; points: Pt[] }[]): string | null {
  const containing = regions
    .filter(r => r.points.length >= 3 && pointInPolygon(pt, r.points))
    .sort((a, b) => polygonArea(a.points) - polygonArea(b.points));
  return containing[0]?.id ?? null;
}

export function regionNearestToPoint(pt: Pt, regions: { id: string; points: Pt[] }[]): string | null {
  let best: { id: string; dist: number } | null = null;
  for (const r of regions) {
    if (r.points.length < 3) continue;
    const { dist } = projectToBoundaryLocal(pt, r.points);
    if (!best || dist < best.dist) best = { id: r.id, dist };
  }
  return best?.id ?? null;
}

import { projectToBoundary as projectToBoundaryLocal } from "./geometry";
```

`src/data/clip.ts`:

```ts
import polygonClipping from "polygon-clipping";
import type { Pt } from "../domain/geometry";
import { isMeaningful, polygonArea } from "../domain/geometry";

type Ring = number[][];

const toRing = (pts: Pt[]): Ring => {
  const r = pts.map(p => [p.x, p.y]);
  const f = r[0], l = r[r.length - 1];
  if (f[0] !== l[0] || f[1] !== l[1]) r.push([f[0], f[1]]);
  return r;
};

const fromGeom = (geom: polygonClipping.Geom | null): Pt[][] => {
  if (!geom) return [];
  return (geom as Ring[][])
    .map(poly => poly[0]
      .filter(_ => true)
      .slice(0, -1)
      .map(([x, y]) => ({ x, y })))
    .filter(pts => pts.length >= 3 && isMeaningful(pts))
    .sort((a, b) => polygonArea(b) - polygonArea(a));
};

export function fitGap(drawn: Pt[], others: Pt[][]): Pt[] | null {
  if (drawn.length < 3) return null;
  const subject = polygonClipping.difference(toRing(drawn), ...others.map(toRing)) as polygonClipping.Polygon | null;
  const [best] = fromGeom(subject as unknown as polygonClipping.Geom);
  return best ?? null;
}

export function claimRest(region: Pt[], others: Pt[][]): Pt[][] {
  if (region.length < 3) return [];
  const subject = polygonClipping.difference(toRing(region), ...others.map(toRing)) as unknown as polygonClipping.Geom;
  return fromGeom(subject as polygonClipping.Geom);
}
```
